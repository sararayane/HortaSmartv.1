EstadoUmidade classificarUmidade(double umidade) {
  if (umidade < 30) {
    return const EstadoUmidade('Crítico', CoresHortasmart.vermelho, CoresHortasmart.vermelhoClaro);
  }
  if (umidade < 50) {
    return const EstadoUmidade('Atenção', CoresHortasmart.amarelo, CoresHortasmart.amareloClaro);
  }
  return const EstadoUmidade('Adequada', CoresHortasmart.verde, CoresHortasmart.verdeClaro);
}

class IndicadorUmidade extends StatelessWidget {
  const IndicadorUmidade({super.key, required this.umidade, this.mostrarTexto = true, this.tamanho = 10});
  final double umidade;
  final bool mostrarTexto;
  final double tamanho;

  @override
  Widget build(BuildContext context) {
    final estado = classificarUmidade(umidade);
    return Row(mainAxisSize: MainAxisSize.min, children: [
      Container(width: tamanho, height: tamanho, decoration: BoxDecoration(color: estado.cor, shape: BoxShape.circle)),
      if (mostrarTexto) ...[
        const SizedBox(width: 7),
        Text(estado.texto, style: TextStyle(color: estado.cor, fontSize: 12, fontWeight: FontWeight.w700)),
      ],
    ]);
  }
}

      case 55:
        return 'Garoa';
      case 56:
      case 57:
        return 'Garoa congelante';
      case 61:
      case 63:
      case 65:
        return 'Chuva';
      case 66:
      case 67:
        return 'Chuva congelante';
      case 71:
      case 73:

  }
}

import 'dart:convert';

import 'package:http/http.dart' as http;

import '../modelos/previsao_tempo.dart';

class ServicoClima {
  static const Duration _tempoLimite = Duration(seconds: 15);

  Future<PrevisaoTempo> buscarPorCidade(String nomeCidade) async {
    final cidade = nomeCidade.trim();

    if (cidade.isEmpty) {
      throw ArgumentError('Digite o nome de uma cidade.');
    }

    final uri = Uri.https(
      'geocoding-api.open-meteo.com',
      '/v1/search',
      {
        'name': cidade,
        'count': '1',
        'language': 'pt',
        'format': 'json',
      },
    );

    final resposta = await http.get(uri).timeout(_tempoLimite);

    if (resposta.statusCode != 200) {
      throw Exception('Não foi possível localizar a cidade.');
    }

    final dados = jsonDecode(resposta.body) as Map<String, dynamic>;
    final resultados = dados['results'] as List<dynamic>?;

    if (resultados == null || resultados.isEmpty) {
      throw Exception(
        'Cidade não encontrada. Confira o nome e tente novamente.',
      );
    }

    final local = resultados.first as Map<String, dynamic>;

    return _buscarPrevisao(
      latitude: (local['latitude'] as num).toDouble(),
      longitude: (local['longitude'] as num).toDouble(),
      cidade: local['name'] as String? ?? cidade,
      pais: local['country'] as String? ?? '',
    );
  }

  Future<PrevisaoTempo> buscarPorCoordenadas({
    required double latitude,
    required double longitude,
  }) async {
    // A consulta meteorológica não depende de encontrar o nome da cidade.
    var cidade = 'Minha localização';
    var pais = '';

    try {
      final uriLocal = Uri.https(
        'nominatim.openstreetmap.org',
        '/reverse',
        {
          'lat': latitude.toString(),
          'lon': longitude.toString(),
          'format': 'jsonv2',
          'accept-language': 'pt-BR',
          'zoom': '10',
        },
      );

      final respostaLocal = await http.get(
        uriLocal,
        headers: {
          'User-Agent': 'HortaSmart/1.0 (aplicativo meteorologico)',
        },
      ).timeout(_tempoLimite);

      if (respostaLocal.statusCode == 200) {
        final dadosLocal =
            jsonDecode(respostaLocal.body) as Map<String, dynamic>;

        final endereco = dadosLocal['address'] as Map<String, dynamic>?;

        if (endereco != null) {
          cidade = endereco['city'] as String? ??
              endereco['town'] as String? ??
              endereco['village'] as String? ??
              endereco['municipality'] as String? ??
              endereco['county'] as String? ??
              cidade;

          pais = endereco['country'] as String? ?? '';
        }
      }
    } catch (_) {
      // Se o nome da cidade não puder ser identificado,
      // a consulta do clima ainda poderá continuar.
    }

    return _buscarPrevisao(
      latitude: latitude,
      longitude: longitude,
      cidade: cidade,
      pais: pais,
    );
  }

  Future<PrevisaoTempo> _buscarPrevisao({
    required double latitude,
    required double longitude,
    required String cidade,
    required String pais,
  }) async {
    final uri = Uri.https(
      'api.open-meteo.com',
      '/v1/forecast',
      {
        'latitude': latitude.toString(),
        'longitude': longitude.toString(),
        'current':
            'temperature_2m,relative_humidity_2m,precipitation,weather_code,wind_speed_10m',
        'hourly': 'precipitation_probability',
        'forecast_days': '1',
        'timezone': 'auto',
      },
    );

    final resposta = await http.get(uri).timeout(_tempoLimite);

    if (resposta.statusCode != 200) {
      throw Exception('Não foi possível consultar a previsão do tempo.');
    }

    final dados = jsonDecode(resposta.body) as Map<String, dynamic>;
    final atual = dados['current'] as Map<String, dynamic>;

    final horario = DateTime.parse(atual['time'] as String);

    final dadosHorarios = dados['hourly'] as Map<String, dynamic>;
    final horarios = dadosHorarios['time'] as List<dynamic>;
    final probabilidades =
        dadosHorarios['precipitation_probability'] as List<dynamic>;

    final horarioAtual = (atual['time'] as String).substring(0, 13);

    var indiceHorario = horarios.indexWhere(
      (item) => (item as String).startsWith(horarioAtual),
    );

    double probabilidadeChuva = 0;

    if (indiceHorario >= 0 && indiceHorario < probabilidades.length) {
      final valor = probabilidades[indiceHorario];

      if (valor != null) {
        probabilidadeChuva = (valor as num).toDouble();
      }
    }

    return PrevisaoTempo(
      cidade: cidade,
      pais: pais,
      temperatura: (atual['temperature_2m'] as num).toDouble(),
      umidadeRelativa: (atual['relative_humidity_2m'] as num).toDouble(),
      probabilidadeChuva: probabilidadeChuva,
      velocidadeVento: (atual['wind_speed_10m'] as num).toDouble(),
      codigoClima: (atual['weather_code'] as num).toInt(),
      atualizadoEm: horario,
    );
  }
}

class TelaIrrigacao extends StatefulWidget {
  const TelaIrrigacao({
    super.key,
    required this.canteiros,
    required this.aoAtualizarCanteiro,
    this.canteiroInicialId,
  });

  final List<Canteiro> canteiros;
  final ValueChanged<Canteiro> aoAtualizarCanteiro;
  final String? canteiroInicialId;

  @override
  State<TelaIrrigacao> createState() => _TelaIrrigacaoState();
}

class _TelaIrrigacaoState extends State<TelaIrrigacao> {
  String? _idSelecionado;
  double _metaTemporaria = 65;
  int _intervaloTemporario = 4;

  Canteiro? get _selecionado {
    if (widget.canteiros.isEmpty) return null;

    for (final canteiro in widget.canteiros) {
      if (canteiro.id == _idSelecionado) {
        return canteiro;
      }
    }

    return widget.canteiros.first;
  }

  @override
  void initState() {
    super.initState();

    if (widget.canteiros.isNotEmpty) {
      Canteiro inicial = widget.canteiros.first;

      for (final item in widget.canteiros) {
        if (item.id == widget.canteiroInicialId) {
          inicial = item;
          break;
        }
      }

      _idSelecionado = inicial.id;
      _carregarConfiguracao(inicial);
    }
  }

  @override
  void didUpdateWidget(covariant TelaIrrigacao oldWidget) {
    super.didUpdateWidget(oldWidget);

    if (widget.canteiros.isEmpty) {
      _idSelecionado = null;
      return;
    }

    if (widget.canteiroInicialId != oldWidget.canteiroInicialId &&
        widget.canteiroInicialId != null) {
      for (final item in widget.canteiros) {
        if (item.id == widget.canteiroInicialId) {
          _idSelecionado = item.id;
          _carregarConfiguracao(item);
          return;
        }
      }
    }

    final selecaoExiste = widget.canteiros.any(
      (canteiro) => canteiro.id == _idSelecionado,
    );

    if (!selecaoExiste) {
      _idSelecionado = widget.canteiros.first.id;
      _carregarConfiguracao(widget.canteiros.first);
    }
  }

  void _carregarConfiguracao(Canteiro canteiro) {
    _metaTemporaria = canteiro.metaUmidade;
    _intervaloTemporario = canteiro.intervaloHoras;
  }

  void _selecionar(Canteiro canteiro) {
    setState(() {
      _idSelecionado = canteiro.id;
      _carregarConfiguracao(canteiro);
    });
  }

  void _mensagem(String texto) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(texto),
        behavior: SnackBarBehavior.floating,
      ),
    );
  }

  void _salvar(Canteiro canteiro) {
    widget.aoAtualizarCanteiro(
      canteiro.copiarCom(
        metaUmidade: _metaTemporaria,
        intervaloHoras: _intervaloTemporario,
      ),
    );

    _mensagem('Configurações salvas nesta prévia.');
  }

  void _alternarBomba(Canteiro canteiro, bool ligar) {
    widget.aoAtualizarCanteiro(
      canteiro.copiarCom(irrigando: ligar),
    );

    _mensagem(
      ligar
          ? 'Irrigação iniciada apenas na prévia. Nenhuma bomba física foi acionada.'
          : 'Irrigação parada apenas na prévia.',
    );
  }

  @override
  Widget build(BuildContext context) {
    if (widget.canteiros.isEmpty) {
      return const AreaPagina(
        child: Center(
          child: Padding(
            padding: EdgeInsets.all(28),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Icon(
                  Icons.water_drop_outlined,
                  size: 42,
                  color: CoresHortasmart.verde,
                ),
                SizedBox(height: 12),

  final ServicoClima _servico = ServicoClima();
  final TextEditingController _cidadeController = TextEditingController();

  PrevisaoTempo? _previsao;
  bool _carregando = false;
  bool _buscandoLocalizacao = false;
  String? _erro;

  Future<void> _buscarClima() async {
    FocusScope.of(context).unfocus();

    setState(() {
      _carregando = true;
      _erro = null;
    });

    try {
      final resultado = await _servico.buscarPorCidade(
        _cidadeController.text,
      );

      if (!mounted) return;

      setState(() {
        _previsao = resultado;
        _cidadeController.text = resultado.cidade;
      });
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _erro = _mensagemErro(e);
      });
    } finally {
      if (mounted) {
        setState(() => _carregando = false);
      }
    }
  }

  Future<void> _buscarPelaLocalizacao() async {
    FocusScope.of(context).unfocus();

    setState(() {
      _buscandoLocalizacao = true;
      _erro = null;
    });

    try {
      final servicosAtivos = await Geolocator.isLocationServiceEnabled();

      if (!servicosAtivos) {
        throw Exception(
          'Ative a localização do dispositivo e tente novamente.',
        );
      }

      var permissao = await Geolocator.checkPermission();

      if (permissao == LocationPermission.denied) {
        permissao = await Geolocator.requestPermission();
      }

      if (permissao == LocationPermission.denied) {
        throw Exception(
          'A permissão de localização foi negada. Autorize para continuar.',
        );
      }

      if (permissao == LocationPermission.deniedForever) {
        throw Exception(
          'A localização está bloqueada nas permissões do sistema. '
          'Ative-a nas configurações do dispositivo.',
        );
      }

      final posicao = await Geolocator.getCurrentPosition(
        locationSettings: const LocationSettings(
          accuracy: LocationAccuracy.medium,
          timeLimit: Duration(seconds: 20),
        ),
      );

      if (!mounted) return;

      setState(() => _carregando = true);

      final resultado = await _servico.buscarPorCoordenadas(
        latitude: posicao.latitude,
        longitude: posicao.longitude,
      );

      if (!mounted) return;

      setState(() {
        _previsao = resultado;
        _cidadeController.text = resultado.cidade;
      });
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _erro = _mensagemErro(e);
      });
    } finally {
      if (mounted) {
        setState(() {
          _carregando = false;
          _buscandoLocalizacao = false;
        });
      }
    }
  }

  IconData _iconeClima(int codigo) {
    if (codigo == 0) return Icons.wb_sunny_rounded;
    if (codigo == 1 || codigo == 2) return Icons.cloud_queue_rounded;
    if (codigo == 3) return Icons.cloud_rounded;

    if (codigo == 45 || codigo == 48) {
      return Icons.foggy;
    }

    if ((codigo >= 51 && codigo <= 67) || (codigo >= 80 && codigo <= 82)) {
      return Icons.grain_rounded;
    }

    if (codigo >= 71 && codigo <= 77 || codigo == 85 || codigo == 86) {
      return Icons.ac_unit_rounded;
    }

    if (codigo >= 95) return Icons.thunderstorm_rounded;

    return Icons.cloud_rounded;
  }
